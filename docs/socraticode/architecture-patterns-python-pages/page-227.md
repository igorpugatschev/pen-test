Здесь мы установили как явные ограничения, такие как «нельзя разме-
щать заказ на большее количество товара, чем имеется в наличии», так
и неявные вроде «каждая товарная позиция заказа размещается в одной
партии товара».

def test_allocating_to_a_batch_reduces_the_available_quantity():
 batch = Batch("batch-001", "SMALL-TABLE", qty=20, eta=date.today())
 line = OrderLine('order-ref', "SMALL-TABLE", 2)
 batch.allocate(line)
 assert batch.available_quantity == 18
